cmake_minimum_required(VERSION 3.16)

project(SpeechToText LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

option(SPEECHTOTEXT_BUILD_EXAMPLES "Build SpeechToText example executable" ON)

set(SPEECHTOTEXT_WHISPER_INCLUDE_DIR "" CACHE PATH "Directory containing whisper.h")
set(SPEECHTOTEXT_WHISPER_LIBRARY "" CACHE FILEPATH "Path to the whisper library")
set(SPEECHTOTEXT_RTAUDIO_INCLUDE_DIR "" CACHE PATH "Directory containing RtAudio.h")
set(SPEECHTOTEXT_RTAUDIO_LIBRARY "" CACHE FILEPATH "Path to the RtAudio library")

find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(PC_WHISPER QUIET whisper)
  pkg_check_modules(PC_RTAUDIO QUIET rtaudio)
endif()

if(SPEECHTOTEXT_WHISPER_INCLUDE_DIR)
  set(WHISPER_INCLUDE_DIR "${SPEECHTOTEXT_WHISPER_INCLUDE_DIR}")
else()
  find_path(WHISPER_INCLUDE_DIR whisper.h HINTS ${PC_WHISPER_INCLUDE_DIRS})
endif()

if(SPEECHTOTEXT_WHISPER_LIBRARY)
  set(WHISPER_LIBRARY "${SPEECHTOTEXT_WHISPER_LIBRARY}")
else()
  find_library(WHISPER_LIBRARY NAMES whisper HINTS ${PC_WHISPER_LIBRARY_DIRS})
endif()

if(SPEECHTOTEXT_RTAUDIO_INCLUDE_DIR)
  set(RTAUDIO_INCLUDE_DIR "${SPEECHTOTEXT_RTAUDIO_INCLUDE_DIR}")
else()
  find_path(RTAUDIO_INCLUDE_DIR RtAudio.h HINTS ${PC_RTAUDIO_INCLUDE_DIRS} PATH_SUFFIXES rtaudio)
endif()

if(SPEECHTOTEXT_RTAUDIO_LIBRARY)
  set(RTAUDIO_LIBRARY "${SPEECHTOTEXT_RTAUDIO_LIBRARY}")
else()
  find_library(RTAUDIO_LIBRARY NAMES rtaudio RtAudio HINTS ${PC_RTAUDIO_LIBRARY_DIRS})
endif()

if(NOT WHISPER_INCLUDE_DIR OR NOT WHISPER_LIBRARY)
  message(FATAL_ERROR "whisper.cpp was not found. Set SPEECHTOTEXT_WHISPER_INCLUDE_DIR and SPEECHTOTEXT_WHISPER_LIBRARY.")
endif()

if(NOT RTAUDIO_INCLUDE_DIR OR NOT RTAUDIO_LIBRARY)
  message(FATAL_ERROR "RtAudio was not found. Set SPEECHTOTEXT_RTAUDIO_INCLUDE_DIR and SPEECHTOTEXT_RTAUDIO_LIBRARY.")
endif()

add_library(speech_to_text STATIC stt_whisper.cpp)
target_compile_features(speech_to_text PUBLIC cxx_std_17)
target_include_directories(
  speech_to_text
  PUBLIC
    "${CMAKE_CURRENT_LIST_DIR}"
    "${WHISPER_INCLUDE_DIR}"
    "${RTAUDIO_INCLUDE_DIR}")
target_link_libraries(speech_to_text PUBLIC "${WHISPER_LIBRARY}" "${RTAUDIO_LIBRARY}")

if(APPLE)
  target_link_libraries(speech_to_text PUBLIC "-framework CoreAudio" "-framework CoreFoundation" "-framework AudioToolbox")
endif()

if(SPEECHTOTEXT_BUILD_EXAMPLES)
  add_executable(example_stt example_stt.cpp)
  set_target_properties(example_stt PROPERTIES OUTPUT_NAME "example_stt.bin")
  target_link_libraries(example_stt PRIVATE speech_to_text)
endif()
